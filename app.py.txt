# 3. محرك النمذجة الرياضية وحساب التنبيهات الشرطية
def analyze_asset(ticker_symbol, asset_name, is_fund=False):
    if not is_fund:
        try:
            stock = yf.Ticker(ticker_symbol)
            df = stock.history(period="6m")
            if df.empty: return None
            
            latest_price = df['Close'].iloc[-1]
            df['MA50'] = df['Close'].rolling(window=50).mean()
            fair_market_price = df['MA50'].iloc[-1] if not np.isnan(df['MA50'].iloc[-1]) else latest_price
            
            delta = df['Close'].diff()
            gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
            loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
            rs = gain / loss
            rsi = 100 - (100 / (1 + rs))
            latest_rsi = rsi.iloc[-1] if not np.isnan(rsi.iloc[-1]) else 50
        except:
            return None
    else:
        latest_price = 100.0  
        fair_market_price = 110.0
        latest_rsi = 45

    # حساب حدود التنبيهات بناءً على النسب الدقيقة المطلوبة من السعر السوقي العادل
    buy_threshold_price = fair_market_price * 0.80   # انخفاض بنسبة 20%
    sell_threshold_price = fair_market_price * 1.15  # ارتفاع بنسبة 15%
    
    alert_status = "HOLD"
    alert_message = ""
    
    if latest_price <= buy_threshold_price:
        alert_status = "BUY"
        alert_message = f"🚀 إشارة شراء واقتناص: سعر {asset_name} الحالي ({latest_price:.2f} ج.م) انخفض بنسبة 20% أو أكثر عن السعر السوقي العادل ({fair_market_price:.2f} ج.م)."
    elif latest_price >= sell_threshold_price:
        alert_status = "SELL"
        alert_message = f"⚠️ إشارة بيع وجني أرباح: سعر {asset_name} الحالي ({latest_price:.2f} ج.م) ارتفع بنسبة 15% أو أكثر عن السعر السوقي العادل ({fair_market_price:.2f} ج.م)."

    return {
        "name": asset_name, "price": latest_price, "fair_price": fair_market_price,
        "rsi": latest_rsi, "alert": alert_status, "message": alert_message
    }
